# Resource Allocator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a minimal hosted Python/FastAPI Resource Allocator that uses realistic synthetic data, validates the activity board, expands activities into task instances, schedules them with explainable traces, and renders readable calendar/audit views.

**Architecture:** Start from committed JSON fixtures so the app works without OpenAI at runtime. Implement schemas, validation, task expansion, simple greedy scheduling, decision traces, and file-backed UI first. Add OpenAI regeneration after the committed-data path is stable.

**Tech Stack:** Python 3.11+, FastAPI, Pydantic, Jinja2, pytest, JSON fixtures, optional OpenAI API regeneration pipeline.

Seed data requirements:

- `activity_families.json` must flatten to at least 100 scheduler-facing activities.
- At least 60 activities must be primary activities.
- Every activity maps to exactly one of `fitness`, `food`, `medication`, `therapy`, `consultation`.
- Include the 8 demo scenarios from the spec.
- Include planned monthly travel and at least one last-minute travel window.
- Include explicit substitutions in the same family as their primary activity.
- Include food prep dependencies and at least one chef/cook provider.
- Use IDs consistently across files.

Per-file content:

- `member_profile.json`: one member profile with goals, preferences, constraints, baseline metrics, scheduling rules, `journey_phases`, ordinary location rhythm, and travel windows.
- `resource_universe.json`: providers, equipment, locations, travel windows, and travel-time rules. Include trainer, physiotherapist, dietitian, physician, phlebotomist/lab, chef/cook, home, office, gym, clinic, lab, travel hotel, remote, and an office-to-gym travel-time rule.
- `known_frictions.json`: planned travel with good facilities, last-minute travel with limited facilities, provider mismatch, lab disruption, food prep dependency, equipment/facility unavailability, travel-time conflict, and skipped high-load recovery adjustment. Give each friction a stable ID.
- `activity_families.json`: families with `activity_family_id`, `intent`, `goal_tags`, `primary_activity`, `substitution_activities`, `substitution_rules`, and `family_validation_notes`. Every activity must include priority, frequency, duration, load, resource references, allowed locations, dependencies, substitutions, metrics, care handoff fields, journey applicability, and `is_primary`.
- `availability.json`: `planning_start_date`, `planning_months`, and `availability_blocks` for member blocks, providers, chef/cook prep windows, equipment, lab, clinic/facilities, planned travel, and last-minute travel.

```python
class ActivityType(StrEnum):

    type: str
    must_happen: str = "before"

    prep_duration_minutes: int | None = None
    provider_type: str | None = None
    can_be_done_by_member: bool = False
    can_be_done_by_provider: bool = False
    depends_on_activity_type: str | None = None

    required_equipment_ids: list[str] = Field(default_factory=list)
    allowed_locations: list[str] = Field(default_factory=list)
    remote_allowed: bool = False
    prep_required: bool = False
    dependencies: list[ActivityDependency] = Field(default_factory=list)

    intent: str
    goal_tags: list[str]
    primary_activity: ActivityPrescription
    substitution_activities: list[ActivityPrescription] = Field(default_factory=list)
    substitution_rules: list[dict[str, str]] = Field(default_factory=list)

    def flattened_activities(self) -> list[ActivityPrescription]:
        return [self.primary_activity, *self.substitution_activities]
```

```python
import json
from pathlib import Path
from typing import TypeVar

from pydantic import BaseModel


T = TypeVar("T", bound=BaseModel)


def read_json(path: str | Path) -> dict | list:
    with Path(path).open("r", encoding="utf-8") as file:
        return json.load(file)


def write_json(path: str | Path, data: dict | list) -> None:
    target = Path(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("w", encoding="utf-8") as file:
        json.dump(data, file, indent=2, sort_keys=True)
```

```python
@dataclass
class ValidationReport:
    passed: bool = True
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    counts: dict[str, int] = field(default_factory=dict)

    def fail(self, message: str) -> None:
        self.passed = False
        self.errors.append(message)


def validate_dataset(
    member: MemberProfile,
    resources: ResourceUniverse,
    families: list[ActivityFamily],
    action_plan: list[ActivityPrescription],
    availability: AvailabilityData,
) -> ValidationReport:
    report = ValidationReport()
    activity_ids = [activity.activity_id for activity in action_plan]
    primary_count = sum(1 for activity in action_plan if activity.is_primary)
    modality_counts = Counter(activity.activity_type.value for activity in action_plan)

    report.counts = {
        "activities": len(action_plan),
        "primary_activities": primary_count,
        "activity_families": len(families),
        "providers": len(resources.providers),
        "equipment": len(resources.equipment),
        "locations": len(resources.locations),
        "availability_blocks": len(availability.availability_blocks),
    }

    if len(action_plan) < 100:
        report.fail("action_plan.json must contain at least 100 scheduler-facing activities.")
    if primary_count < 60:
        report.fail("action_plan.json must contain at least 60 primary activities.")
    if len(activity_ids) != len(set(activity_ids)):
        report.fail("Activity IDs must be unique.")

    required_modalities = {"fitness", "food", "medication", "therapy", "consultation"}
    missing_modalities = required_modalities - set(modality_counts)
    if missing_modalities:
        report.fail(f"Missing required modalities: {sorted(missing_modalities)}")

    provider_ids = resources.provider_ids()
    equipment_ids = resources.equipment_ids()
    location_ids = resources.location_ids()
    activity_id_set = set(activity_ids)

    for activity in action_plan:
        for provider_id in activity.required_provider_ids:
            if provider_id not in provider_ids:
                report.fail(f"{activity.activity_id} references unknown provider {provider_id}.")
        for equipment_id in activity.required_equipment_ids:
            if equipment_id not in equipment_ids:
                report.fail(f"{activity.activity_id} references unknown equipment {equipment_id}.")
        for location_id in activity.allowed_locations:
            if location_id not in location_ids and location_id != "remote":
                report.fail(f"{activity.activity_id} references unknown location {location_id}.")
        for substitution_id in activity.substitution_activity_ids:
            if substitution_id not in activity_id_set:
                report.fail(f"{activity.activity_id} references unknown substitution {substitution_id}.")

    if not member.journey_phases:
        report.fail("Member profile must include journey_phases.")

    return report
```

```python
from datetime import date, timedelta

from src.models.activity import ActivityPrescription
from src.models.schedule import TaskInstance


def week_key(day: date) -> str:
    iso_year, iso_week, _ = day.isocalendar()
    return f"{iso_year}-W{iso_week:02d}"


def expand_activity(activity: ActivityPrescription, start: date, end: date) -> list[TaskInstance]:
    frequency = activity.frequency
    frequency_type = frequency.get("type")
    instances: list[TaskInstance] = []
    cursor = start
    index = 1

    if frequency_type == "daily":
        step = 1
        count_per_week = None
    elif frequency_type == "weekly":
        step = 7
        count_per_week = int(frequency.get("count", 1))
    elif frequency_type == "monthly":
        step = 30
        count_per_week = None
    else:
        step = 7
        count_per_week = 1

    while cursor <= end:
        repetitions = count_per_week or 1
        for offset in range(repetitions):
            target = cursor + timedelta(days=offset * max(1, 6 // max(1, repetitions)))
            if target > end:
                continue
            instances.append(TaskInstance(
                task_instance_id=f"task_{activity.activity_id}_{index:03d}",
                activity_id=activity.activity_id,
                activity_family_id=activity.activity_family_id,
                is_substitution=not activity.is_primary,
                target_date=target.isoformat(),
                target_week=week_key(target),
                duration_minutes=activity.duration_minutes,
                priority=activity.priority,
                goal_tags=activity.goal_tags,
                required_resources=[*activity.required_provider_ids, *activity.required_equipment_ids],
                dependencies=[dependency.model_dump(mode="json") for dependency in activity.dependencies],
                status="pending",
            ))
            index += 1
        cursor += timedelta(days=step)
    return instances


def expand_action_plan(action_plan: list[ActivityPrescription], start: date, end: date) -> list[TaskInstance]:
    primary_activities = [activity for activity in action_plan if activity.is_primary]
    instances: list[TaskInstance] = []
    for activity in primary_activities:
        instances.extend(expand_activity(activity, start, end))
    return instances
```

```python
def check_no_member_overlap(candidate: dict, existing_rows: list[CalendarRow]) -> ConstraintCheck:
    candidate_date = candidate["date"]
    start = candidate["start_time"]
    end = candidate["end_time"]
    for row in existing_rows:
        if row.date != candidate_date:
            continue
        if start < row.end_time and end > row.start_time:
            return ConstraintCheck(name="member_overlap", passed=False, reason="Candidate overlaps an existing member event.")
    return ConstraintCheck(name="member_overlap", passed=True, reason="No member overlap.")


def check_same_day_repeat(activity: ActivityPrescription, candidate: dict, existing_rows: list[CalendarRow]) -> ConstraintCheck:
    if activity.same_day_repeat_allowed:
        return ConstraintCheck(name="same_day_repeat", passed=True, reason="Same-day repeats allowed.")
    for row in existing_rows:
        if row.date == candidate["date"] and row.title == activity.title:
            return ConstraintCheck(name="same_day_repeat", passed=False, reason="Same activity already scheduled today.")
    return ConstraintCheck(name="same_day_repeat", passed=True, reason="No same-day repeat.")
```

```python
def resources_available(candidate: dict, required_resource_ids: list[str], availability: AvailabilityData) -> ConstraintCheck:
    if not required_resource_ids:
        return ConstraintCheck(name="resource_availability", passed=True, reason="No concrete resources required.")

    for block in availability.availability_blocks:
        if block.start <= candidate["start"] and block.end >= candidate["end"]:
            available_ids.add(block.resource_id)
    missing = [resource_id for resource_id in required_resource_ids if resource_id not in available_ids]
    if missing:
        return ConstraintCheck(name="resource_availability", passed=False, reason=f"Missing available resources: {missing}")
    return ConstraintCheck(name="resource_availability", passed=True, reason="All required resources available.")
```

```python
from datetime import date, datetime, time, timedelta

from src.models.activity import ActivityPrescription
from src.models.availability import AvailabilityData
from src.models.schedule import CalendarRow, TaskInstance
from src.models.trace import DecisionTrace
from src.scheduler.policy import check_no_member_overlap, check_same_day_repeat

DEFAULT_TIMES = [time(8, 0), time(12, 30), time(17, 30), time(19, 0)]


def make_candidate(day: date, start_time: time, duration_minutes: int, location_id: str) -> dict:
    start_dt = datetime.combine(day, start_time)
    end_dt = start_dt + timedelta(minutes=duration_minutes)
    return {
        "date": day.isoformat(),
        "start_time": start_dt.strftime("%H:%M"),
        "end_time": end_dt.strftime("%H:%M"),
        "start": start_dt.isoformat(),
        "end": end_dt.isoformat(),
        "location_id": location_id,
    }


def schedule_tasks(
    action_by_id: dict[str, ActivityPrescription],
    task_instances: list[TaskInstance],
    availability: AvailabilityData,
) -> tuple[list[CalendarRow], list[DecisionTrace]]:
    rows: list[CalendarRow] = []
    traces: list[DecisionTrace] = []
    sorted_tasks = sorted(task_instances, key=lambda task: (task.priority, task.target_date, task.task_instance_id))

    for task in sorted_tasks:
        activity = action_by_id[task.activity_id]
        selected = None
        selected_checks = []
        target_day = date.fromisoformat(task.target_date)
        locations = activity.allowed_locations or ["home"]
        for location_id in locations:
            for candidate_time in DEFAULT_TIMES:
                candidate = make_candidate(target_day, candidate_time, activity.duration_minutes, location_id)
                checks = [
                    check_no_member_overlap(candidate, rows),
                    check_same_day_repeat(activity, candidate, rows),
                ]
                dependency_checks = []
                if activity.dependencies:
                    dependency_checks.append(
                        build_dependency_acknowledgement_check(activity.activity_id)
                    )
                if all(check.passed for check in checks + dependency_checks):
                    selected = candidate
                    selected_checks = checks
                    break
            if selected:
                break

        trace_id = f"trace_{task.task_instance_id}"
        if selected:
            row = CalendarRow(
                calendar_row_id=f"row_{task.task_instance_id}",
                date=selected["date"],
                start_time=selected["start_time"],
                end_time=selected["end_time"],
                title=activity.title,
                activity_type=activity.activity_type.value,
                goal_tags=activity.goal_tags,
                load_level=activity.load_level,
                location_id=selected["location_id"],
                mode="remote" if selected["location_id"] == "remote" else "in_person",
                substitution_status="substitution" if task.is_substitution else "primary",
                trace_id=trace_id,
            )
            rows.append(row)
            traces.append(build_trace(trace_id, task.task_instance_id, task.activity_id, "scheduled", selected, selected_checks, [], build_provider_handoff(activity)))
        else:
            traces.append(build_trace(trace_id, task.task_instance_id, task.activity_id, "unscheduled", None, selected_checks, [], build_provider_handoff(activity)))

    return rows, traces


def build_dependency_acknowledgement_check(activity_id: str):
    from src.models.trace import ConstraintCheck

    return ConstraintCheck(name="dependency_feasibility", passed=True, reason=f"Dependencies for {activity_id} are acknowledged for MVP scheduling.")


def build_provider_handoff(activity: ActivityPrescription) -> str | None:
    if not activity.share_with_provider_types:
        return None
    context = ", ".join(activity.care_context_required) if activity.care_context_required else "standard context"
    providers = ", ".join(activity.share_with_provider_types)
    return f"Share {context} with {providers}."
```

```python
from pathlib import Path


def render_prompt(fragment_paths: list[str], context: dict[str, str]) -> str:
    parts = []
    for fragment_path in fragment_paths:
        text = Path(fragment_path).read_text(encoding="utf-8")
        for key, value in context.items():
            text = text.replace("{{ " + key + " }}", value)
        parts.append(text)
    return "\n\n".join(parts)
```
